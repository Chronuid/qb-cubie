import json
from pathlib import Path


def merge_packages(source, destination, priority=None, schema=None):
    paths=sorted(Path(source).glob('*/*.json'),key=lambda path:path.relative_to(source).as_posix())
    ranks={name:index for index,name in enumerate(priority or [])}
    paths.sort(key=lambda path:(ranks.get(path.parent.name,-1),path.relative_to(source).as_posix()))
    merged={}
    for path in paths:
        merged.setdefault(path.name,{}).update(json.loads(path.read_text(encoding='utf-8')))
    if schema is not None and any(set(values)-set(schema) for values in merged.values()):
        raise ValueError('unknown schema key')
    output=Path(destination)
    output.mkdir()
    for name,values in merged.items():
        (output/name).write_text(json.dumps(values,sort_keys=True,separators=(',',':'))+'\n',encoding='utf-8')
