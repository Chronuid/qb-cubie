from pathlib import Path
import pytest
from document_catalog import build_catalog

DATA = [{'inputs': {'z.txt': b'last\x00\xff', 'a.txt': b'first', 'data.csv': b'a,b\n', 'sub/deep.txt': b'nested', 'private/no.txt': b'private', 'index.txt': b'input index'}, 'expected': {'index.txt': b'a.txt\ndata.csv\nindex.txt\nz.txt\n', 'files/a.txt': b'first', 'files/data.csv': b'a,b\n', 'files/index.txt': b'input index', 'files/z.txt': b'last\x00\xff'}}, {'inputs': {}, 'expected': {'index.txt': b''}}]


def observation(root, item):
    source_root=root/'inputs'
    source_root.mkdir()
    for name,content in item['inputs'].items():
        path=source_root/name
        path.parent.mkdir(parents=True,exist_ok=True)
        path.write_bytes(content)
    source=source_root
    output=root/'out'
    try:
        returned=build_catalog(source,output)
    except NotImplementedError:
        returned='unimplemented'
    tree={p.relative_to(output).as_posix():p.read_bytes() for p in output.rglob('*') if p.is_file()}
    inputs={p.relative_to(source_root).as_posix():p.read_bytes() for p in source_root.rglob('*') if p.is_file()}
    return {'tree':tree,'returned':returned,'inputs':inputs}

@pytest.fixture(params=DATA)
def observed(tmp_path, request):
    item=request.param
    return observation(tmp_path,item),item

def test_copies(observed):
    actual,item=observed
    assert {name:data for name,data in actual['tree'].items() if name!='index.txt'}=={name:data for name,data in item['expected'].items() if name!='index.txt'}

def test_sorted_index(observed):
    actual,item=observed
    assert actual['tree'].get('index.txt')==item['expected']['index.txt']

def test_exact_tree(observed):
    actual,item=observed
    assert actual['tree']==item['expected']

def test_returns_none(observed):
    actual,item=observed
    assert actual['returned'] is None

def test_preserve_inputs(observed):
    actual,item=observed
    assert actual['inputs']==item['inputs']
